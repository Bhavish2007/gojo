Program 01

Chi-Squared Test for Association

# Step1: Load the data and organize it into data frames

data <- data.frame(
  Price = c(3,2,1,3,2,1,3,2,1,3,2,1,3,2,1),
  Rating = c(0,0,0,1,1,1,2,2,2,3,3,3,4,4,4),
  Respondents = c(4,1,4,8,21,20,26,22,23,15,12,9,7,4,4)
)
data

# Step2: Generate a cross contingency table summing up total respondent for each
# price-rating combination.

table_data <- xtabs(Respondents ~ Price + Rating, data = data)

table_data

# Step3: Create a multi-category bar plot to visualize the data
barplot(table_data, legend=T, xlab="Ratings", ylab="No. of Respondents")

# Step 4: Run chi-squared test to test the Hypothesis
chisq.test(table_data)

#Step 5: Monte-Carlo simulation
chisq.test(table_data,simulate.p.value=TRUE, B=2500)
Program 02

One-Way ANOVA

#Step 1: load the CSV file


#Step 2: visualization of mean using Boxplot

boxplot(Score~Teaching_Method,data=teaching_method_scores,col = "lightblue", main =
"Distribution of Test Scores by Teaching Method", xlab = "Teaching Method", ylab = "Test
Score")

#Step 3: One way ANOVA

anova_result <- aov(Score ~ Teaching_Method, data = teaching_method_scores)

summary(anova_result)


#Step 4: Tukey HSD post hoc testing

tukey_results <- TukeyHSD(anova_result)

print(tukey_results)
Program 03

Two-Way ANOVA

#Step 1: load the CSV file and organize it into data frames.

# Step 2: Using box plot, visualizing light vs temperature for different glass types.
boxplot(Light ~ Temp * Glass, data = oscilloscope_light_output, col = c("lightblue",
"lightgreen"), main = "Boxplot of Light vs Temperature for Different Glass Types",xlab =
"Temperature",ylab = "Light")

# Step 3: Formulate a hypothesis about the effect of glass type and temperature on light
# output. Run Two way ANOVA.

anova_result= aov(Light ~ Glass * Temp, data = oscilloscope_light_output)

summary(anova_result)

# Step 4: Conduct Post-hoc testing

TukeyHSD(anova_result)
